cmake_minimum_required(VERSION 3.16)
# Standalone local prototype checks, or the normal server test suite.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(RadioNoisePrototype LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 14)
    find_package(Qt5 REQUIRED COMPONENTS Core Test)
    find_package(Threads REQUIRED)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(OPUS REQUIRED IMPORTED_TARGET opus)
    get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
    add_library(audio-feedback STATIC "${ROOT}/src/murmur/RadioNoise.cpp" "${ROOT}/src/murmur/AudioSignal.cpp"
        "${ROOT}/src/murmur/AudioFeedback.cpp" "${ROOT}/src/murmur/SquelchPlan.cpp")
    target_include_directories(audio-feedback PUBLIC "${ROOT}/src/murmur")
    target_link_libraries(audio-feedback PUBLIC Qt5::Core Threads::Threads PkgConfig::OPUS)
    enable_testing()
    add_executable(TestNoiseScheduler ../TestNoiseScheduler/TestNoiseScheduler.cpp)
    target_link_libraries(TestNoiseScheduler PRIVATE audio-feedback)
    target_compile_definitions(TestNoiseScheduler PRIVATE AUDIO_ASSET_DIR="${ROOT}/assets/audio")
    add_test(NAME TestNoiseScheduler COMMAND TestNoiseScheduler)
    foreach(test TestAudioFeedback TestAudioVolume TestBusyToneAsset)
        add_executable(${test} "../${test}/${test}.cpp" "${ROOT}/src/murmur/SimplexController.cpp")
        set_target_properties(${test} PROPERTIES AUTOMOC ON)
        target_link_libraries(${test} PRIVATE audio-feedback Qt5::Test)
        target_compile_definitions(${test} PRIVATE AUDIO_ASSET_DIR="${ROOT}/assets/audio" BUSY_ASSET_DIR="${ROOT}/assets/audio")
        add_test(NAME ${test} COMMAND ${test})
    endforeach()
endif()
add_executable(TestRadioNoise TestRadioNoise.cpp)
target_link_libraries(TestRadioNoise PRIVATE audio-feedback)
add_test(NAME TestRadioNoise COMMAND TestRadioNoise)
set_tests_properties(TestRadioNoise PROPERTIES TIMEOUT 120)
